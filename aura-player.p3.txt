let audioCtx=null,oscNodes=[];
function startTone(){
  stopTone();
  try{
    audioCtx=audioCtx||new(window.AudioContext||window.webkitAudioContext)();
    if(audioCtx.state==="suspended")audioCtx.resume();
    const master=audioCtx.createGain();master.gain.value=0.04;master.connect(audioCtx.destination);
    const freqs=[110,164.8,220,329.6];
    oscNodes=freqs.map((f,i)=>{
      const o=audioCtx.createOscillator(),g=audioCtx.createGain();
      o.type=i%2?"sine":"triangle";o.frequency.value=f;g.gain.value=0.15/(i+1);
      o.connect(g);g.connect(master);o.start();return o;
    });
  }catch(_){}
}
function stopTone(){oscNodes.forEach(o=>{try{o.stop()}catch(_){}});oscNodes=[];}
